65 struct timeout_observer_strategy
66 {
67 static constexpr auto preferred_disposables_mode = rpp::details::observers::disposables_mode::None;
68
69 std::shared_ptr<timeout_disposable<TObserver, TFallbackObservable, Container>> disposable;
70
71 void set_upstream(const rpp::disposable_wrapper& d) const
72 {
73 disposable->add(d);
74 }
75
76 bool is_disposed() const
77 {
78 return disposable->is_disposed();
79 }
80
81 template<typename T>
82 void on_next(T&& v) const
83 {
84 auto obs_with_timeout = disposable->get_observer_with_timeout_under_lock();
85 // the timer may have disposed and handed the observer to the fallback while we waited for the lock
86 if (disposable->is_disposed())
87 return;
88 obs_with_timeout->observer.on_next(std::forward<T>(v));
89 obs_with_timeout->timeout = rpp::schedulers::utils::get_worker_t<TScheduler>::now() + disposable->get_period();
90 }
91
92 void on_error(const std::exception_ptr& err) const noexcept
93 {
94 const auto obs_with_timeout = disposable->get_observer_with_timeout_under_lock();
95 if (!disposable->is_disposed())
96 obs_with_timeout->observer.on_error(err);
97 }
98
99 void on_completed() const noexcept
100 {
101 const auto obs_with_timeout = disposable->get_observer_with_timeout_under_lock();
102 if (!disposable->is_disposed())
103 obs_with_timeout->observer.on_completed();
104 }
105 };

108 struct timeout_t
109 {
110 template<rpp::constraint::decayed_type T>
111 struct operator_traits
112 {
113 static_assert(rpp::constraint::observable_of_type<TFallbackObservable, T>, "TFallbackObservable should be the same type as T");
114
115 using result_type = T;
116
117 constexpr static bool own_current_queue = true;
118 };

120 template<rpp::details::observables::constraint::disposables_strategy Prev>
121 using updated_optimal_disposables_strategy = rpp::details::observables::fixed_disposables_strategy<1>;
122
123 rpp::schedulers::duration period;
124 RPP_NO_UNIQUE_ADDRESS TFallbackObservable fallback;
125 RPP_NO_UNIQUE_ADDRESS TScheduler scheduler;
126
127 template<rpp::constraint::decayed_type Type, rpp::details::observables::constraint::disposables_strategy DisposableStrategy, rpp::constraint::observer Observer>
128 auto lift_with_disposables_strategy(Observer&& observer) const
129 {
130 using worker_t = rpp::schedulers::utils::get_worker_t<TScheduler>;
131 using container = typename DisposableStrategy::disposables_container;
132
133 const auto timeout = worker_t::now() + period;
134
135 const auto disposable = disposable_wrapper_impl<timeout_disposable<std::decay_t<Observer>, TFallbackObservable, container>>::make(std::forward<Observer>(observer), period, fallback, timeout);
136 auto ptr = disposable.lock();
137 ptr->get_observer_with_timeout_under_lock()->observer.set_upstream(disposable.as_weak());
138
139 const auto worker = scheduler.create_worker();
140 using wrapper = timeout_disposable_wrapper<std::decay_t<Observer>, TFallbackObservable, container>;
141 worker.schedule(
142 timeout,
143 [](wrapper& handler) -> rpp::schedulers::optional_delay_to {
144 auto locked_obs_with_timeout = handler.disposable->get_observer_with_timeout_under_lock();
145 if (rpp::schedulers::utils::get_worker_t<TScheduler>::now() < locked_obs_with_timeout->timeout)
146 return rpp::schedulers::delay_to(locked_obs_with_timeout->timeout);
147
148 if (!handler.disposable->is_disposed())
149 {
150 handler.disposable->dispose();
151 handler.disposable->get_fallback().subscribe(std::move(locked_obs_with_timeout->observer));
152 }
153 return std::nullopt;
154 },
155 wrapper{ptr});
156
157 return rpp::observer<Type, timeout_observer_strategy<std::decay_t<Observer>, TFallbackObservable, container, TScheduler>>{std::move(ptr)};
158 }
159 };
